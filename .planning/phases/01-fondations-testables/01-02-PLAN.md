```python
# src/deskgate/core/paths.py — aucun import de sqlite3/subprocess/httpx2/fastapi/ctypes/uvicorn
Resolver = Callable[[str], str]

@dataclass(frozen=True)
class Mapping:
    container: str   # ex. "/data"
    host: str        # ex. "C:/BIM"

class PathDenied(Exception):
    def __init__(self, code: str, message: str) -> None: ...
    code: str        # "path_invalid" | "path_outside_roots"
    message: str     # français

def default_resolver(p: str) -> str          # str(Path(p).resolve(strict=False))
def normalize_host_path(raw: str) -> str     # lexical hôte, renvoie "C:/BIM/x" (séparateurs "/", lettre de lecteur majuscule)
def split_container_path(raw: str) -> list[str]  # lexical conteneur, renvoie les segments ; "/data" -> ["data"]
def is_under(path: str, root: str) -> bool   # lexical, par segments, insensible à la casse ; la racine elle-même => True
def container_to_host(raw: str, mappings: Sequence[Mapping]) -> str
def host_to_container(host_path: str, mappings: Sequence[Mapping]) -> str
def resolve_within_roots(host_path: str, roots: Sequence[str], resolver: Resolver = default_resolver) -> str
def validate_container_path(raw: str, mappings: Sequence[Mapping], roots: Sequence[str], resolver: Resolver = default_resolver) -> str
def validate_artifact_path(host_path: str, mappings: Sequence[Mapping], roots: Sequence[str], resolver: Resolver = default_resolver) -> str
```
