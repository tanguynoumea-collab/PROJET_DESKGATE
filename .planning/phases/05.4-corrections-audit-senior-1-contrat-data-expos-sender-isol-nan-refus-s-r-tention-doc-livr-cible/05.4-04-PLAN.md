```python
_PURGED_DATA = '{"purged":true}'


def purge_data(conn: sqlite3.Connection, *, older_than: datetime, now: datetime | None = None) -> int:
    """Remplace `$.data` par {"purged": true} (DS-SCAL-1) ; renvoie le nombre de tâches purgées.

    Seulement : tâche terminale, terminée avant `older_than`, rappel dans un état final, `data` objet non vide, ni
    déjà purgé ni marqueur de repli (`truncated`, qui pointe `data.json`). Une ligne au JSON invalide est ignorée.
    `summary`, `artifacts`, `callbacks` et l'audit existant ne sont pas touchés ; un événement `data.purged` est
    ajouté si au moins une tâche est purgée. Aucune migration : le fichier ne rétrécit pas (pas de VACUUM).
    """
    finals = sorted(CALLBACK_FINAL_STATES)
    marks = ",".join("?" * len(finals))
    moment = _now(now)
    cutoff = iso(older_than)
    with tx(conn):
        cur = conn.execute(
            "UPDATE jobs SET result_json = json_set(result_json, '$.data', json(?)) "
            "WHERE result_json IS NOT NULL AND finished_at IS NOT NULL AND finished_at < ? "
            "AND status IN ('succeeded','failed','timeout','cancelled','interrupted') "
            f"AND id IN (SELECT job_id FROM callbacks WHERE state IN ({marks})) "
            "AND CASE WHEN json_valid(result_json) THEN "
            "json_type(result_json, '$.data') = 'object' "
            "AND json_extract(result_json, '$.data') NOT IN ('{}', ?) "
            "AND json_type(result_json, '$.data.truncated') IS NOT 'true' "
            "ELSE 0 END",
            (_PURGED_DATA, cutoff, *finals, _PURGED_DATA),
        )
        count = cur.rowcount
        if count > 0:
            _audit(conn, "data.purged", None, {"count": count, "older_than": cutoff}, moment)
    return count
```
Le `CASE WHEN json_valid(...)` est obligatoire (Pitfall 6 de la recherche) : SQLite ne garantit pas l'ordre
d'évaluation d'un `AND`, et `json_type`/`json_extract` lèvent « malformed JSON » sur une ligne invalide ; le `CASE`
garantit que `json_valid` est évalué d'abord. Aucun changement à `MIGRATIONS` ni à `SCHEMA_VERSION`. Le f-string ne
contient que des `?` générés (aucune donnée interpolée). Si `ruff` signale S608 sur cette ligne, ajouter
`# noqa: S608` avec le commentaire « marqueurs générés, aucune valeur interpolée ».

```python
@app.command()
def purge(
    older_than_days: Annotated[
        int,
        typer.Option(
            "--older-than-days",
            min=0,
            help="Purge data des tâches terminées depuis plus de N jours (rappel final). Défaut : 30.",
        ),
    ] = 30,
) -> None:
    """Retire `data` des tâches terminées dont le rappel est final (summary et artefacts conservés)."""
    path = db_path(resolve_home())
    if not path.is_file():
        _err(f"Base introuvable : {path}. Rien à purger.")
        raise typer.Exit(1)
    cutoff = datetime.now(UTC) - timedelta(days=older_than_days)
    conn = store.connect(path)
    try:
        version = int(conn.execute("PRAGMA user_version").fetchone()[0])
        if version > store.SCHEMA_VERSION:
            _err(str(store.SchemaTooNew(version)))
            raise typer.Exit(2)
        if version == 0:
            _err("Base non initialisée : lancez une fois `deskgate serve`. Rien à purger.")
            raise typer.Exit(2)
        count = store.purge_data(conn, older_than=cutoff)
    except store.OperationalError as exc:
        _err(f"Base occupée ou illisible ({type(exc).__name__}) : réessayez dans quelques secondes.")
        raise typer.Exit(1) from exc
    finally:
        conn.close()
    typer.echo(
        f"{count} tâche(s) purgée(s) : data des tâches terminées avant le "
        f"{cutoff:%Y-%m-%d %H:%M} UTC, rappel final. Le fichier de base ne rétrécit pas."
    )
```
