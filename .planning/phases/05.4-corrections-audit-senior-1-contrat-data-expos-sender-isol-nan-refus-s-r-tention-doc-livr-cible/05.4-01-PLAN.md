```python
def _schedule(self, claim: store.CallbackClaim, after: datetime, *, short: bool) -> tuple[str, float, datetime | None]:
    """État suivant d'une ligne non finale : pending avec échéance, ou abandoned_exhausted hors fenêtre."""
    delay = next_delay(claim.attempt, max_backoff_s=self._config.callback.max_backoff_s, rng=self._rng, short=short)
    if after + timedelta(seconds=delay) >= claim.deadline:
        return ABANDONED_EXHAUSTED, delay, None  # pas de dernier envoi hors fenêtre
    return "pending", delay, after + timedelta(seconds=delay)
```

def _record_internal_error(self, conn: store.Connection, claim: store.CallbackClaim, exc: Exception) -> None:
    name = type(exc).__name__

    after = self._wall()
