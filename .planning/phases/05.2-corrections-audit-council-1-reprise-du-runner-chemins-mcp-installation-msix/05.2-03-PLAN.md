```python
EXIT_THREAD_FAILED = 6
MAX_RESTARTS = 3
WINDOW_S = 600.0

@dataclass(frozen=True)
class Decision:
    restart_engine: bool
    restart_sender: bool
    give_up: bool

def recent(times: Sequence[float], now: float, window: float = WINDOW_S) -> list[float]: ...
def decide(*, engine_failed: bool, sender_failed: bool, recent_restarts: int,
           max_restarts: int = MAX_RESTARTS) -> Decision: ...
def exit_code(*, gave_up: bool) -> int: ...   # 0 ou EXIT_THREAD_FAILED

class Restartable(Protocol):
    failed: bool
    on_fatal: Callable[[], None] | None
    def start(self) -> None: ...
    def stop(self, timeout: float = 5.0) -> None: ...
    def wake(self) -> None: ...

@dataclass
class Holder:
    engine: Restartable
    sender: Restartable

class Supervisor:
    def __init__(self, *, holder: Holder, make_engine: Callable[[], Restartable],
                 make_sender: Callable[[], Restartable], recover: Callable[[], list[str]],
                 on_give_up: Callable[[], None], clock: Callable[[], float] = time.monotonic,
                 period: float = 1.0, max_restarts: int = MAX_RESTARTS, window_s: float = WINDOW_S) -> None: ...
    gave_up: bool
    last_cause: str | None
    def notify(self) -> None: ...
    def attach(self) -> None: ...      # pose on_fatal = self.notify sur holder.engine et holder.sender
    def check(self) -> None: ...       # un passage, testable sans thread
    def start(self) -> None: ...       # thread "deskgate-supervisor", daemon
    def stop(self, timeout: float = 5.0) -> None: ...
```
