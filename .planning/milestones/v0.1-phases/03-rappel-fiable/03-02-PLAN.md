```python
DELIVERED = "delivered"; DELIVERED_DUPLICATE = "delivered_duplicate"

@dataclass(frozen=True)
```

    ```python
    def _message(body: bytes | str | None) -> str:
        if not body:
            return ""
        text = body[:BODY_READ_MAX].decode("utf-8", "replace") if isinstance(body, bytes) else body[:BODY_READ_MAX]
        try:
            data = json.loads(text)
            msg = data.get("message") if isinstance(data, dict) else None
        except ValueError:
            msg = None
        return (msg if isinstance(msg, str) else text[:200]).lower()

    def classify(status, body, *, maybe_delivered, prev_status=None) -> Decision:
        if status is None:
            return Decision(RETRY_BACKOFF, "network")
        if 200 <= status < 300:
            return Decision(DELIVERED)
        if status == 409:
            msg = _message(body)
            if "finished" in msg:
                return Decision(DELIVERED_DUPLICATE if maybe_delivered else ABANDONED_STALE, "finished")
            if "running already" in msg or "about to resume" in msg:
                return Decision(RETRY_SHORT, "running")
            return Decision(RETRY_SHORT, "unknown_409")
        if status in (401, 403, 404):
            if prev_status != status:                       # deux constats consécutifs identiques (STATE.md)
                return Decision(RETRY_SHORT, f"confirm_{status}")
            return Decision(ABANDONED_STALE if status == 404 else ABANDONED_REJECTED, f"http_{status}")
        if status == 413:                                   # un seul renvoi minimal (STATE.md)
            return Decision(RETRY_BACKOFF if prev_status == 413 else RETRY_MINIMAL, "too_large")
        return Decision(RETRY_BACKOFF, f"http_{status}")

    def next_delay(attempts, *, max_backoff_s, rng, short=False) -> float:
        base = 2.0 * (1.5 if short else 2.0) ** min(max(attempts - 1, 0), 30)
        cap = SHORT_CAP_S if short else float(max_backoff_s)
        return min(cap, base * rng.uniform(0.75, 1.25))
    ```
